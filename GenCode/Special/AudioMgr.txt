        private void OnTimers(object param)
        {
            m_removeBnks.Clear();
            var enumator = m_bnkDic.GetEnumerator();
            while (enumator.MoveNext())
            {
                var handler = enumator.Current.Value;
                if (handler.IsUsing)
                {
                    continue;
                }

                if (Time.realtimeSinceStartup - handler.LastSleepTime >= m_unloadBnkDelays)
                {
                    m_removeBnks.Add(enumator.Current.Key);
                }
            }
            enumator.Dispose();
            if (m_removeBnks.Count > 0)
            {
                foreach (var bnk in m_removeBnks)
                {
                    m_bnkDic.Remove(bnk);
                    AkBankManager.UnloadBank(bnk);                    
                }
            }
        }

        private void CheckBnk(string bnkName)
        {
            if (string.IsNullOrEmpty(bnkName))
            {
                return;
            } 
            if (!m_bnkDic.ContainsKey(bnkName))
            {
                var handler = new WwiseBankHandle(bnkName);
                m_bnkDic[bnkName] = handler;
                AkBankManager.LoadBank(bnkName, false, false);
            }
        }

        private void IncreaseBnk(string bnkName,uint playingId,string eventName)
        {
            if (m_bnkDic.TryGetValue(bnkName,out var handler))
            {
                handler.IncreaseRef(playingId,eventName);
            }
        }

        public uint PlayEvent(string eventName, GameObject target,uint flags = (uint)AkCallbackType.AK_EndOfEvent)
        {
#if DISABLE_WISEEVENT
            return 0;
#else
            bool showAudioLog = FrameworkConfig.Inst.Release != ReleaseType.Release && FrameworkConfig.Inst.EnableAudioLog;
            
            WiseEventConfig cfg = WiseEventDao.Inst.GetCfg(eventName);
            if (cfg == null)
            {
                HHL.Common.Log.Inst.Print("PlayEvent cfgName[" + eventName + "] => [WiseEventDao cfg err !!!]",HHL.Common.Log.eLogType.ewise);
            }
            else
            {
                HHL.Common.Log.Inst.Print("PlayEvent cfgName[" + eventName + "] => [" + cfg.WiseEvent+"]",HHL.Common.Log.eLogType.ewise ); 
            }
            
            
            if (cfg == null || string.IsNullOrEmpty(cfg.WiseEvent))
            {
                if (showAudioLog)
                {
                    Logger.LogError($"WiseEventConfig Null Or Empty:{eventName}");
                }
                return 0;
            }

            if (cfg.BgSound == 1 && CheckBgAudioConflict(cfg.OppositeName))
            {
                if (showAudioLog)
                {
                    Logger.Log($"{eventName} BgSound = 1 And Check Bg Audio Conflict:{cfg.OppositeName}");
                }
                return 0;
            }

            if (showAudioLog)
            {
                Logger.Log($"PlayAudioEvent:{eventName},EventName:{cfg.WiseEvent}");
            }
            return PostEvent(cfg.WiseEvent,cfg.Bnk, target,flags);
#endif
        }

        public void PlayEvent(string eventName, GameObject target, Action playEndAct)
        {
#if DISABLE_WISEEVENT
            return ;
#else

#if ENABLE_AUDIO_LOG
            Debug.Log("-----PlayEvent:" + eventName);
#endif
            bool showAudioLog = FrameworkConfig.Inst.Release != ReleaseType.Release && FrameworkConfig.Inst.EnableAudioLog;

            WiseEventConfig cfg = WiseEventDao.Inst.GetCfg(eventName);
            if (cfg == null || string.IsNullOrEmpty(cfg.WiseEvent))
            {
                return;
            }
            if (StopPlayWiseByBnk(cfg.Bnk))
            {
                return;
            }
            if (cfg.BgSound == 1 && CheckBgAudioConflict(cfg.OppositeName))
            {
                return;
            }
            //2025 11 06 空bnk事件不发 wwise请求
            if (string.IsNullOrEmpty(cfg.Bnk))
            {
                if (FrameworkConfig.Inst.Release != ReleaseType.Release && FrameworkConfig.Inst.EnableAudioLog)
                {
                    Logger.LogError($"记录无Bnk事件:EventName:{eventName}");
                }
                return ;
            }

            
#if UNITY_STANDALONE && !UNITY_EDITOR
            if (!AppCache.Common.AppFocus)
            {
                return;
            }
#endif            
            
            CheckBnk(cfg.Bnk);
            
            RecordAudio(cfg.WiseEvent,cfg.Bnk);
            
            uint ret = AkUnitySoundEngine.PostEvent(cfg.WiseEvent, target != null ? target : ListenerObject,
                (uint) AkCallbackType.AK_EndOfEvent, AudioPlayEndEvent, m_akCallPackage);
            m_endEventDic[ret] = AudioEndEventType.OtherFunc;
            m_endActDic[ret] = playEndAct;
            if (ret > 0)
            {
                IncreaseBnk(cfg.Bnk,ret, cfg.WiseEvent);
                m_playIdEvents[ret] = cfg.Bnk;    
            }
            
            if (showAudioLog)
            {
                Logger.Log($"PlayAudioEvent:{eventName},EventName:{cfg.WiseEvent}");
            }
#endif
        }

        private bool CheckBgAudioConflict(string newBgAudio)
        {
            if (string.IsNullOrEmpty(m_curBgAudioEvent))
            {
                m_preBgAudioEvent = string.Empty;
                m_curBgAudioEvent = newBgAudio;
                m_conflictBgAudioEvent = string.Empty;
                return false;
            }

            if (m_curBgAudioEvent.Equals(newBgAudio))
            {
                return false;
            }

            if (m_bgAudioDelDict.TryGetValue(m_curBgAudioEvent, out var checkDel))
            {
                if (checkDel())
                {
                    m_conflictBgAudioEvent = newBgAudio;
                    return true;
                }
            }

            m_preBgAudioEvent = m_curBgAudioEvent;
            m_curBgAudioEvent = newBgAudio;
            m_conflictBgAudioEvent = string.Empty;
            return false;
        }

        public void PlayPreConflictAudioEvent()
        {
            if (string.IsNullOrEmpty(m_conflictBgAudioEvent))
            {
                if (!string.IsNullOrEmpty(m_preBgAudioEvent))
                {
                    PlayEvent(m_preBgAudioEvent);
                }
            }
            else
            {
                PlayEvent(m_conflictBgAudioEvent);
            }
        }
        

        private AkCallbackManager.EventCallbackPackage m_akCallPackage = new AkCallbackManager.EventCallbackPackage();

        /// <summary>
        /// 播放 英雄 语音，附带转对讲机音效
        /// </summary>
        /// <param name="eventName"></param>
        /// <returns></returns>
        public uint PlayInterPhoneEvent(string eventName, GameObject target = null)
        {
#if DISABLE_WISEEVENT
            return 0;
#else
            WiseEventConfig cfg = WiseEventDao.Inst.GetCfg(eventName);
            if (cfg == null || string.IsNullOrEmpty(cfg.WiseEvent))
            {
                return 0;
            }

            if (StopPlayWiseByBnk(cfg.Bnk))
            {
                return 0;
            }
            //2025 11 06 空bnk事件不发 wwise请求
            if (string.IsNullOrEmpty(cfg.Bnk))
            {
                if (FrameworkConfig.Inst.Release != ReleaseType.Release && FrameworkConfig.Inst.EnableAudioLog)
                {
                    Logger.LogError($"记录无Bnk事件:EventName:{cfg.WiseEvent}");
                }
                return 0;
            }

#if UNITY_STANDALONE && !UNITY_EDITOR
            if (!AppCache.Common.AppFocus)
            {
                return 0;
            }
#endif             
            //增加对讲机事件
            bool playPhoneEvent = false;
            WiseEventConfig phoneEventCfg = WiseEventDao.Inst.GetCfg("interphone_open");
            if (phoneEventCfg != null && !string.IsNullOrEmpty(phoneEventCfg.WiseEvent))
            {
                PostEvent(phoneEventCfg.WiseEvent, cfg.Bnk,null);
                playPhoneEvent = true;
            }

            if (playPhoneEvent && !m_playInterPhoneEvent)
            {
                CheckBnk(cfg.Bnk);
                m_playInterPhoneEvent = true;

                RecordAudio(cfg.WiseEvent, cfg.Bnk);
                    
                uint ret = AkUnitySoundEngine.PostEvent(cfg.WiseEvent, target == null ? ListenerObject : target,
                    (uint) AkCallbackType.AK_EndOfEvent, AudioPlayEndEvent, m_akCallPackage);
                m_endEventDic.Add(ret, AudioEndEventType.StopInterPhone);

                if (ret > 0)
                {
                    IncreaseBnk(cfg.Bnk,ret,cfg.WiseEvent);
                    m_playIdEvents[ret] = cfg.Bnk;
                }
                
                return ret;
            }
            else
            {
                return PostEvent(cfg.WiseEvent, cfg.Bnk,null);
            }
#endif
        }

        /// <summary>
        /// 当配置中bnk为空时，release包不发事件，非release包正常发事件供调试
        /// </summary>

        private bool StopPlayWiseByBnk(string bnk)

            }

        }

        private void AudioPlayEndEvent(object in_cookie, AkCallbackType in_type, AkCallbackInfo in_info)
        {
            if (in_type == AkCallbackType.AK_EndOfEvent)
            {
                var info = in_info as AkEventCallbackInfo;
                if (info == null)
                {
                    return;
                }
                uint playId = info.playingID;
                if (m_endEventDic.TryGetValue(playId, out AudioEndEventType type))
                {
                    if (type == AudioEndEventType.StopInterPhone)
                    {
                        m_playInterPhoneEvent = false;
                        PlayEvent("interphone_close");
                    }
                    else if (type == AudioEndEventType.OtherFunc)
                    {
                        if (m_endActDic.TryGetValue(playId, out Action endAct))
                        {
                            endAct.Invoke();
                            m_endActDic.Remove(playId);
                        }
                    }
                    m_endEventDic.Remove(playId);
                }
                
                if (m_playIdEvents.TryGetValue(playId, out var bnk))
                { 
                    if (m_bnkDic.TryGetValue(bnk, out var handler))
                    {
                        handler.DecreaseRef(playId);
                    }
                    else
                    {
                        Logger.LogWarning($"end of event,bnk:{bnk},but has no exist handler");
                    }
                }
            }
        }

        public uint PostEvent(string eventName,string bnk, GameObject target,uint flags = (uint)AkCallbackType.AK_EndOfEvent)
        {
#if DISABLE_WISEEVENT
            return 0;
#else
            if (StopPlayWiseByBnk(bnk))
            {
                return 0;
            }

            
#if UNITY_STANDALONE && !UNITY_EDITOR
            if (!AppCache.Common.AppFocus)
            {
                return 0;
            }
#endif
            //2025 11 06 空bnk事件不发 wwise请求
            if (string.IsNullOrEmpty(bnk))
            {
                if (FrameworkConfig.Inst.Release != ReleaseType.Release && FrameworkConfig.Inst.EnableAudioLog)
                {
                    Logger.LogError($"记录无Bnk事件:EventName:{eventName}");
                }
                return 0;
            }
            
            CheckBnk(bnk);
            GameObject go = target != null ? target : ListenerObject;
            
            RecordAudio(eventName,bnk);
            
            uint ret = AkUnitySoundEngine.PostEvent(eventName, go,flags, AudioPlayEndEvent,null);
            if (ret > 0)
            {
                IncreaseBnk(bnk,ret,eventName);
                m_playIdEvents[ret] = bnk;
            }
            
            return ret;
#endif
        }

    public class WwiseBankHandle
    {
        private readonly string m_bnkName;
        public int RefCount { get; private set; }
        public float LastSleepTime { get; private set; }
        public bool IsUsing {get { return RefCount > 0; } }

        private Dictionary<uint, string> m_infoDict = new Dictionary<uint, string>();
        public WwiseBankHandle(string bnkName)
        {
            m_bnkName = bnkName;
            RefCount = 0;
        }

        public void IncreaseRef(uint id,string eventName)
        {
            m_infoDict[id] = eventName;
            RefCount++;
        }

        public void DecreaseRef(uint id)
        {
            if (m_infoDict.ContainsKey(id))
            {
                m_infoDict.Remove(id);
            }
             
            RefCount--;
            if (RefCount == 0)
            {
                LastSleepTime = Time.realtimeSinceStartup;
            }
            else if (RefCount < 0)
            {
                Logger.LogWarning($"{m_bnkName} RefCount < 0");
            }
        }
        
    }
